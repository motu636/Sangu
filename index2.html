static int search(int[] a, int t) {
  int lo = 0, hi = a.length - 1;
  while (lo <= hi) {
    int mid = (lo + hi) >>> 1;
    if (a[mid] == t) return mid;
    if (a[mid] < t) lo = mid + 1;
    else hi = mid - 1;
  }
  return -1;
}

def top_words(text, n=3):
    words = text.lower().split()
    return Counter(words).most_common(n)

        head.next, prev, head = prev, head, head.next
    return prev
